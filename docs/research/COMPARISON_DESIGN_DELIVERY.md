  --parent data/interim/connectivity-falsification/fil-connectivity-falsification-1/376e928980ff1d0d3ef3a220af212ac679919036 \
  --pipeline-revision b5842b696814715e0e8500597ca3d02a6a2adfd2

uv run --frozen python scripts/verify_complete_corpus.py LOCAL_CORPUS_BUNDLE LOCAL_CONFIG

uv build
